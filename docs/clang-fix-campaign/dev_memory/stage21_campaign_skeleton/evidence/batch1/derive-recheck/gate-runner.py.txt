"""Record an unmodified full pytest run and its load envelope."""

import datetime
import json
import os
import subprocess
import sys
import time
import tomllib
import xml.etree.ElementTree as ET
from pathlib import Path

root = Path.cwd()
output = Path(sys.argv[1]).resolve()
output.mkdir(parents=True, exist_ok=False)


def write(name, data):
    (output / name).write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n")


def load():
    return {"utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "loadavg": Path("/proc/loadavg").read_text().strip(),
            "logical_cpu_count": os.cpu_count()}


env = os.environ.copy()
where = tomllib.loads((root / "pyproject.toml").read_text())["tool"]["setuptools"]["packages"]["find"]["where"]
env["PYTHONPATH"] = os.pathsep.join(["/tmp", *(str(root / p) for p in where)])
env["PYTHONDONTWRITEBYTECODE"] = "1"
env["SKEL_GATE_FAILURES"] = str(output / "failures.jsonl")
env["P5Q_C5A_EVIDENCE"] = str(output / "c5a")
env["P5Q_C8_INVESTIGATION"] = str(output / "c8")
env["P5Q_C6_FIX_EVIDENCE"] = str(output / "c6-fix")
for key in ("QB_COOKIE", "QB_PASSWORD", "DEBUG", "PWDEBUG", "NODE_OPTIONS"):
    env.pop(key, None)
before = load()
write("load-before.json", before)
top = subprocess.run(["ps", "-eo", "comm=,pcpu=", "--sort=-pcpu"], capture_output=True, text=True, check=True)
(output / "cpu-top10.txt").write_text("\n".join(top.stdout.splitlines()[:10]) + "\n")
argv = [str(root / ".venv/bin/python"), "tests/run_quiet.py", "-q", "-p", "skel_b1_gate_watch",
        "--junitxml=" + str(output / "pytest.xml"), "-o", "cache_dir=" + str(output / "pytest-cache")]
write("command.json", {"argv": argv, "cwd": str(root), "start_utc": before["utc"],
                        "gate_observer": "/tmp/skel_b1_gate_watch.py", "test_edits": False})
started = time.monotonic()
print("START " + str(output), flush=True)
with (output / "pytest.stdout.log").open("w") as stdout, (output / "pytest.stderr.log").open("w") as stderr:
    run = subprocess.run(argv, env=env, stdout=stdout, stderr=stderr)
after = load()
write("load-after.json", after)
summary = {"exit": run.returncode, "start_utc": before["utc"], "end_utc": after["utc"],
           "seconds": time.monotonic() - started}
if (output / "pytest.xml").exists():
    suites = ET.parse(output / "pytest.xml").getroot()
    summary["suites"] = [s.attrib for s in suites.iter("testsuite")]
    summary["failures"] = [{"classname": t.get("classname"), "name": t.get("name")}
                           for t in suites.iter("testcase") if t.find("failure") is not None or t.find("error") is not None]
write("summary.json", summary)
print(json.dumps(summary), flush=True)
sys.exit(run.returncode)
