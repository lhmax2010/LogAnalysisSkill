import datetime
import json
import os
from pathlib import Path
import subprocess
import sys
import time

ROOT = Path('/home/linhao/Toolchain/development/LogAnalysisSkill')
DEST = ROOT / 'docs/clang-fix-campaign/dev_memory/stage21_campaign_skeleton/evidence/batch2/implementation'
DEST.mkdir(parents=True, exist_ok=True)
PY = str(ROOT / '.venv/bin/python')

def load():
    return dict(loadavg=Path('/proc/loadavg').read_text().strip(), cpu_count=os.cpu_count())

def run(name, argv):
    folder = DEST / name
    folder.mkdir(exist_ok=False)
    before = load()
    before['top_cpu_processes'] = subprocess.check_output(
        ['ps', '-eo', 'comm,pcpu', '--sort=-pcpu'], text=True).splitlines()[:11]
    start = datetime.datetime.now(datetime.timezone.utc).isoformat()
    tick = time.monotonic()
    env = os.environ.copy()
    if name.startswith('timing-'):
        env['P5Q_C5A_EVIDENCE'] = str(folder/'node')
    with (folder / 'stdout.log').open('w') as out, (folder / 'stderr.log').open('w') as err:
        result = subprocess.run(argv, cwd=ROOT, stdout=out, stderr=err, env=env)
    record = dict(argv=argv, cwd=str(ROOT), started=start,
        finished=datetime.datetime.now(datetime.timezone.utc).isoformat(),
        seconds=time.monotonic()-tick, exit_code=result.returncode,
        load_before=before, load_after=load())
    (folder / 'command.json').write_text(json.dumps(record, indent=2)+'\n')
    print(name, result.returncode, round(record['seconds'], 2), flush=True)
    print((folder / 'stdout.log').read_text()[-2400:], flush=True)
    if result.returncode:
        print((folder / 'stderr.log').read_text()[-2400:], flush=True)
    return result.returncode
