"""Local-only selection evidence; no campaign, network, or build commands."""

import csv
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import time
from datetime import datetime, timezone
import xml.etree.ElementTree as ET

REPO = Path.cwd()
OUT = Path(__file__).resolve().parent
HOME = Path('/home/linhao')
BUNDLE = HOME / 'qb_analysis/1204735_vs_1202044'
SUCCESS = HOME / 'Toolchain/development/qb_1203397_lld/work'
PACKAGES = ('zlib', 'libtirpc', 'elfutils')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def capture(name, argv):
    folder = OUT / name
    folder.mkdir(parents=True, exist_ok=False)
    start = datetime.now(timezone.utc).isoformat()
    tick = time.monotonic()
    env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1')
    with (folder / 'stdout.txt').open('wb') as stdout, (folder / 'stderr.txt').open('wb') as stderr:
        result = subprocess.run(argv, stdout=stdout, stderr=stderr, env=env, check=False)
    record = {'argv': argv, 'started_utc': start,
              'ended_utc': datetime.now(timezone.utc).isoformat(),
              'seconds': time.monotonic() - tick, 'exit': result.returncode}
    (folder / 'command.json').write_text(json.dumps(record, indent=2) + '\n')
    return record


commands = [
    ('free', ['free', '-h']),
    ('swap', ['swapon', '--show']),
    ('load', ['cat', '/proc/loadavg']),
    ('nproc', ['nproc']),
    ('df', ['df', '-h', str(REPO), str(HOME / 'GBS-ROOT-TIZEN-UNIFIED-LLVM'),
            str(HOME / 'GBS-ROOT-TOOLCHAIN-LLVM-R4-20260714'),
            str(REPO / 'tmp/GBS-ROOT-TIZEN-UNIFIED-LLVM')]),
    ('vmstat', ['vmstat', '1', '3']),
]
environment = {name: capture('environment-raw/' + name, argv) for name, argv in commands}
# ps exposes only names and numeric RSS, never arguments or environment.
ps = subprocess.run(['ps', '-eo', 'comm=,rss=', '--sort=-rss'],
                    capture_output=True, text=True, check=True)
top = ps.stdout.splitlines()[:15]
(OUT / 'environment-raw/top-rss.txt').write_text('COMMAND RSS_KiB\n' + '\n'.join(top) + '\n')
environment['top_rss'] = {'argv': ['ps', '-eo', 'comm=,rss=', '--sort=-rss'],
                          'first_rows': 15, 'rows': top}
(OUT / 'environment-raw/meminfo.txt').write_text(Path('/proc/meminfo').read_text())
write('environment-commands.json', environment)

manifest = HOME / 'Downloads/tizen-base-toolchain_20261020.074527_standard.xml'
members = {p.attrib['name']: p.attrib for p in ET.parse(manifest).getroot().findall('project')}
status = json.loads((SUCCESS / 'status.json').read_text())
with (BUNDLE / 'llvm_1204735/packages/failed_packages.tsv').open() as stream:
    failures = list(csv.DictReader(stream, delimiter='\t'))
facts = []
for package in PACKAGES:
    log = BUNDLE / f'llvm_1204735/packages/standard/x86_64/failed/{package}.buildlog.txt'
    before = sha(log)
    record = capture('analyzer/' + package,
                     [str(REPO / '.venv/bin/python'), '-B', '-m', 'gbs_analyzer', 'analyze',
                      str(log), '--output-format', 'both', '--max-tokens', '1800',
                      '--output-dir', str(OUT / 'analyzer' / package / 'output')])
    assert sha(log) == before
    elapsed = []
    for arch in ('armv7l', 'aarch64', 'x86_64'):
        success_log = SUCCESS / 'logs' / arch / (package + '.log')
        boundaries = [{'line': i, 'text': line} for i, line in
                      enumerate(success_log.read_text().splitlines(), 1)
                      if re.search(r'(started|finished) "build ' + package + r'\.spec"', line)]
        elapsed.append({'arch': arch, 'log': str(success_log), 'sha256': sha(success_log),
                        'boundaries': boundaries})
    lines = log.read_text().splitlines()
    evidence = [{'line': i, 'text': line} for i, line in enumerate(lines, 1)
                if any(token in line for token in
                       ('toybox: Unknown command', 'autopoint: ***', 'autoreconf: error:',
                        'Bad exit status', 'failed "build', 'started "build', 'export CC=',
                        'init_buildsystem --configdir'))]
    facts.append({'package': package, 'analyzer': record, 'log': str(log), 'log_sha256': before,
                  'manifest': str(manifest), 'manifest_sha256': sha(manifest),
                  'project': members['platform/upstream/' + package],
                  'failure_rows': [r for r in failures if r['package'] == package],
                  'success_rows': [r for r in status if r['source'] == package],
                  'success_time_evidence': elapsed, 'failure_excerpt': evidence})
write('source-facts.json', facts)
configs = []
for which in ('base', 'effective_base', 'packages'):
    path = BUNDLE / 'llvm_1204735/config' / which / 'build.conf'
    configs.append({'path': str(path), 'sha256': sha(path), 'bytes': path.stat().st_size})
write('config-hashes.json', configs)
write('run.json', {'completed_utc': datetime.now(timezone.utc).isoformat(),
                   'packages': list(PACKAGES), 'analyzer_runs': len(facts),
                   'all_logs_unchanged': True, 'network_commands': 0, 'build_commands': 0,
                   'campaign_commands': 0, 'browser_commands': 0,
                   'note': 'Pre-screening only; none yet qualifies under all selection criteria.'})
print(json.dumps({'analyzer_exits': {r['package']: r['analyzer']['exit'] for r in facts}}))
